```php
#[AsTaggedItem(index: 'resize')]
final class ResizeStepHandler implements StepHandlerInterface {
    public function __construct(
        private HttpClientInterface $embedderClient, // service id: embedder.client
        #[Autowire(param: 'transformations.embedder_timeout_resize_ms')] private int $defaultTimeoutMs,
    ) {}
    public static function supportedType(): StepType { return StepType::RESIZE; }
    public function defaultTimeoutMs(): int { return $this->defaultTimeoutMs; }
    public function run(string $bytes, array $params, int $timeoutMs): HandlerResult {
        $start = (int) (microtime(true) * 1000);
        $response = $this->embedderClient->request('POST', '/img/resize', [
            'timeout' => $timeoutMs / 1000.0,
            'headers' => ['Accept' => 'image/*'],
            'body' => [
                'image' => new DataPart($bytes, 'input', 'application/octet-stream'),
                'params' => json_encode($params, JSON_THROW_ON_ERROR),
            ],
        ]);
        $status = $response->getStatusCode(); // déclenche I/O
        if ($status >= 400) {
            throw new TransformationPipelineException(sprintf('embedder /img/resize → %d : %s', $status, $response->getContent(false)));
        }
        $contentType = $response->getHeaders()['content-type'][0] ?? 'application/octet-stream';
        $renderMs = (int) ($response->getHeaders()['x-render-duration-ms'][0] ?? ((int)(microtime(true)*1000) - $start));
        return new HandlerResult($response->getContent(), $contentType, $renderMs);
    }
}
```

**PipelineRunner** :
```php
final class PipelineRunner {
    /** @var array<string, StepHandlerInterface> */
    private array $handlersByType;

    public function __construct(
        #[AutowireIterator('app.step_handler')] iterable $handlers,
        private int $hardCapMs, // %transformations.hard_cap_ms%
        private LoggerInterface $logger,
    ) {
        foreach ($handlers as $h) {
            $this->handlersByType[$h::supportedType()->value] = $h;
        }
    }

    public function run(AssetTransformation $tx, string $bytes, string $outputExt): PipelineResult {
        $outputExt = strtolower(ltrim($outputExt, '.'));
        $start = (int) (microtime(true) * 1000);
        $applied = [];
        $contentType = 'application/octet-stream';

        // Snapshot des steps triés
        $steps = $tx->getSteps()->toArray();
        usort($steps, fn($a,$b) => $a->getSortOrder() <=> $b->getSortOrder());

        // Append implicit format_convert si extension diffère du dernier format_convert
        $lastFormat = null;
        foreach ($steps as $s) {
            if ($s->getType() === StepType::FORMAT_CONVERT) {
                $lastFormat = strtolower($s->getParams()['format'] ?? '');
            }
        }
        $normalizedRequested = $outputExt === 'jpg' ? 'jpeg' : $outputExt;
        $normalizedLast = $lastFormat === 'jpg' ? 'jpeg' : $lastFormat;
        $needsImplicitConvert = ($normalizedLast !== $normalizedRequested);

        $effectiveSteps = $steps;
        if ($needsImplicitConvert) {
            $effectiveSteps[] = $this->makeVirtualFormatConvertStep($outputExt);
        }

        foreach ($effectiveSteps as $step) {
            $type = $step->getType();
            if (!isset($this->handlersByType[$type->value])) {
                throw new TransformationPipelineException("Unsupported step type: {$type->value}", TransformationPipelineException::CODE_UNSUPPORTED_STEP);
            }
            $handler = $this->handlersByType[$type->value];
            $elapsed = (int)(microtime(true)*1000) - $start;
            $remaining = $this->hardCapMs - $elapsed;
            if ($remaining <= 0) {
                throw new TransformationPipelineException("Hard cap {$this->hardCapMs}ms exceeded before step {$type->value}", TransformationPipelineException::CODE_CAP_EXCEEDED);
            }
            $timeoutMs = min($handler->defaultTimeoutMs(), $remaining);
            try {
                $res = $handler->run($bytes, $step->getParams() ?? [], $timeoutMs);
            } catch (TransportExceptionInterface $e) {
                throw new TransformationPipelineException("Embedder transport error on {$type->value}: ".$e->getMessage(), TransformationPipelineException::CODE_EMBEDDER_ERROR, $e);
            }
            $bytes = $res->bytes;
            $contentType = $res->contentType;
            $applied[] = $type->value;
        }

        $totalMs = (int)(microtime(true)*1000) - $start;
        if ($totalMs > $this->hardCapMs) {
            throw new TransformationPipelineException("Hard cap exceeded (post): {$totalMs}ms", TransformationPipelineException::CODE_CAP_EXCEEDED);
        }

        return new PipelineResult($bytes, $contentType, $totalMs, $tx->getWarnings(), $applied);
    }

    private function makeVirtualFormatConvertStep(string $ext): TransformationStep {
        $s = new TransformationStep();
        $s->setType(StepType::FORMAT_CONVERT);
        $s->setSortOrder(PHP_INT_MAX);
        $s->setParams(['format' => $ext]); // jpg/jpeg/png/webp/avif
        return $s;
    }
}
```
