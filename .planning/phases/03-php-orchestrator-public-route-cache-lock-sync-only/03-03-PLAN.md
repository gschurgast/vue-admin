**Route** : `GET /t/{code}/{id}.{ext}` où :
- `{code}` : `[a-z][a-z0-9-]{1,62}[a-z0-9]` (kebab-case, déjà validé par TRANSFORM-06)
- `{id}` : `\d+`
- `{ext}` : `png|jpg|jpeg|webp|avif` (router requirement → 404 sinon)

**Headers réponse 200** :
- `Cache-Control: public, max-age=31536000, immutable`
- `ETag: "{transformationId}-v{hash8}-{assetId}"` (NON quoted dans la string mais quoted par Symfony)
- `Content-Type: image/{webp|png|jpeg|avif}` (jpg → image/jpeg)
- `Content-Length: <bytes>` quand connu
- `Cross-Origin-Resource-Policy: cross-origin`
- `Vary: Accept` (laissé minimal)
- `X-Transformation-Warnings: alpha-flatten-on-jpeg, ...` si warnings non vides (séparés par `, ` — specifics)

**Headers réponse 503 (cap exceeded ou lock waiter)** : `Retry-After: 2` (seconds).
**Headers réponse 502 (embedder error after retries)** : pas de Retry-After.
**Headers réponse 304** : ETag identique à la 200, sans body, sans Content-Length.
**Headers réponse 404** : `Cache-Control: public, max-age=300` (TTL court pour erreurs publiques).

```php
namespace App\Controller;

final class PublicTransformationController extends AbstractController
{
    public function __construct(
        #[Autowire(param: 'transformations.public_route.enabled')] private bool $enabled,
        private TransformationLookup $lookup,
        private VariantCache $cache,
        private FilesystemOperator $assetsStorage,
        private LockFactory $lockFactory, // service id: lock.transformations.factory
        private PipelineRunner $runner,
        private LoggerInterface $logger,
    ) {}

    public function serve(string $code, int $id, string $ext, Request $request): Response
    {
        if (!$this->enabled) {
            return $this->notFoundShort();
        }
        try {
            [$tx, $asset] = $this->lookup->findOr404($code, $id);
        } catch (NotFoundHttpException) {
            return $this->notFoundShort();
        }

        $extNorm = strtolower($ext);
        $contentType = $this->extToContentType($extNorm); // jpg→image/jpeg etc.
        $storageKey = TransformationStorageKey::forVariant($tx->getId(), $tx->getVersionHash(), $asset->getId(), $extNorm);
        $hash8 = substr($tx->getVersionHash(), 0, 8);
        $etag = sprintf('"%d-v%s-%d-%s"', $tx->getId(), $hash8, $asset->getId(), $extNorm);

        // If-None-Match short-circuit
        if ($request->headers->get('If-None-Match') === $etag) {
            return $this->withCommonHeaders(new Response(null, 304), $etag, $contentType, $tx);
        }

        // Cache hit
        if ($this->cache->has($storageKey)) {
            return $this->streamFromCache($storageKey, $etag, $contentType, $tx);
        }

        // Cache miss → lock
        $lock = $this->lockFactory->createLock('lock:tx:'.$storageKey, ttl: 10.0, autoRelease: true);
        if (!$lock->acquire(false)) {
            // Waiter behavior (D-04) : attendre un peu et revérifier S3
            $deadline = microtime(true) + 5.0; // wait up to 5s
            while (microtime(true) < $deadline) {
                usleep(250_000);
                if ($this->cache->has($storageKey)) {
                    return $this->streamFromCache($storageKey, $etag, $contentType, $tx);
                }
            }
            return new Response('', 503, ['Retry-After' => '2']);
        }

        $lockReleased = false;
        try {
            // Re-check après acquisition (un autre worker a pu finir entre temps).
            // W5 : si cache hit ici → release IMMÉDIATEMENT le lock AVANT de streamer.
            // Le lock couvre la GÉNÉRATION, pas le streaming d'un blob déjà persisté.
            if ($this->cache->has($storageKey)) {
                $lock->release();
                $lockReleased = true;
                return $this->streamFromCache($storageKey, $etag, $contentType, $tx);
            }
            // Charger bytes originaux
            try {
                $originalBytes = $this->assetsStorage->read($asset->getS3Key());
            } catch (\Throwable $e) {
                return $this->notFoundShort(); // asset binaire manquant → 404
            }
            try {
                $result = $this->runner->run($tx, $originalBytes, $extNorm);
            } catch (TransformationPipelineException $e) {
                $this->logger->warning('transformation pipeline failed', ['code'=>$e->getCode(), 'msg'=>$e->getMessage()]);
                $status = match ($e->getCode()) {
                    TransformationPipelineException::CODE_CAP_EXCEEDED => 503,
                    TransformationPipelineException::CODE_EMBEDDER_ERROR => 502,
                    TransformationPipelineException::CODE_UNSUPPORTED_STEP => 404,
                    default => 500,
                };
                $headers = $status === 503 ? ['Retry-After' => '2'] : [];
                return new Response('', $status, $headers);
            }
            $this->cache->write($storageKey, $result->bytes, $result->contentType);
            // W5 : release AVANT le stream (la génération est terminée, S3 contient l'objet).
            $lock->release();
            $lockReleased = true;
            return $this->streamFromCache($storageKey, $etag, $result->contentType, $tx);
        } finally {
            // Garde-fou : si une exception survient avant le release explicite, on libère ici.
            if (!($lockReleased ?? false)) {
                $lock->release();
            }
        }
    }

    private function streamFromCache(string $key, string $etag, string $contentType, AssetTransformation $tx): StreamedResponse
    {
        $response = new StreamedResponse(function() use ($key) {
            $stream = $this->cache->read($key);
            fpassthru($stream);
            if (is_resource($stream)) fclose($stream);
        }, 200);
        return $this->withCommonHeaders($response, $etag, $contentType, $tx);
    }

    private function withCommonHeaders(Response $r, string $etag, string $contentType, AssetTransformation $tx): Response
    {
        $r->headers->set('Cache-Control', 'public, max-age=31536000, immutable');
        $r->headers->set('ETag', $etag);
        $r->headers->set('Content-Type', $contentType);
        $r->headers->set('Cross-Origin-Resource-Policy', 'cross-origin');
        $warnings = $tx->getWarnings();
        if (!empty($warnings)) {
            $codes = array_map(fn($w) => $w['code'] ?? '', $warnings);
            $r->headers->set('X-Transformation-Warnings', implode(', ', array_filter($codes)));
        }
        return $r;
    }

    private function notFoundShort(): Response
    {
        return new Response('', 404, ['Cache-Control' => 'public, max-age=300']);
    }

    private function extToContentType(string $ext): string
    {
        return match ($ext) {
            'jpg', 'jpeg' => 'image/jpeg',
            'png' => 'image/png',
            'webp' => 'image/webp',
            'avif' => 'image/avif',
        };
    }
}
```
